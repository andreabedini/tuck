# Tuck M3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the execution phase: give every artifact an on-disk path, run each stale
action single-threaded in its own pure-Tcl directory sandbox, and expose it as
`tuck build <label>`.

**Architecture:** Builds on M2. Analysis gains `ctx path` (artifact → exec path, a pure
string function), `-private` outputs, and stricter validation in `ctx action` (known
artifacts, own outputs, one producer per artifact, no `exec` redirection words). Two
new root-interpreter modules do the work: `lib/sandbox.tcl` runs one action in a
fresh directory containing only its declared inputs, with a cleared environment;
`lib/execute.tcl` orders the needed actions leaves-first, decides staleness against
an action database, and drives the sandbox. The child interpreters are unchanged.

**Tech Stack:** Tcl 9.0+ (`tclsh`), `tcltest`, Jujutsu (`jj`). Sandbox tests use POSIX
`sh`, `cp`, `cat`, `tr`, `printf`.

**Reference spec:** `docs/superpowers/specs/2026-11-08-tuck-m3-design.md`

## Version control & test conventions

- jj: working copy is a commit; new files auto-tracked; finalize with `jj commit -m "..."`.
- Run one suite: `tclsh tests/<name>.test`; full suite: `tclsh tests/all.tcl`.
- A clean run ends `Total N Passed N Skipped S Failed 0`. `S` is 0 unless `cc` is
  missing, in which case the example build test (Task 10) is skipped.
- Every task ends with the full suite green before committing.

## Background the implementer needs

- Artifacts live in `::tuck::graph::artifacts` (id → dict). Source:
  `{type source path <repo-rel>}`; declared: `{type declared target <label> relpath <rel>}`.
- Actions live in `::tuck::graph::actions` (id → `{id target inputs outputs cmd env dynamic}`),
  ids `<label>#<n>`, insertion order is leaves-first.
- `::tuck::graph::target_outputs` (label → published output ids) feeds `ctx dep_outputs`.
- `::tuck::reset` (in `lib/loader.tcl`) clears every global store; each new store
  must be added there.
- `ctx` accessors are procs `::tuck::ctx::<sub> {ctx args...}` where `ctx` is the
  target node; `::tuck::ctx::dispatch` has an explicit allow-list of subcommands.
- `::tuck::config::root` is the normalized repo root.
- Tcl `exec` treats words beginning with `<`, `>`, `|`, `2>` and a final `&` as
  redirections. Commands that need shell features must be `sh -c {script}`.
- `::env` is a magic array: never `array unset ::env` (it drops the traces). Unset
  elements one by one.

```tcl
# Path of an artifact relative to the exec root (the repo root, or a sandbox).
# Pure string function: no I/O, safe to expose to rule impls via `ctx path`.
proc ::tuck::graph::exec_path {id} {
    if {![dict exists $::tuck::graph::artifacts $id]} {
        error "tuck: unknown artifact '$id'"
    }
    set a [dict get $::tuck::graph::artifacts $id]
    if {[dict get $a type] eq "source"} {
        return [dict get $a path]
    }
    regexp {^//(.*):([^:]+)$} [dict get $a target] -> pkg name
    set base [expr {$pkg eq "" ? "tuck-out" : "tuck-out/$pkg"}]
    return "$base/__${name}__/[dict get $a relpath]"
}
```

```tcl
# Declare an output artifact for this target; returns its id. Published in
# target_outputs (visible to dependents via dep_outputs) unless -private.
proc ::tuck::ctx::declare_output {ctx relpath args} {
    set private 0
    foreach opt $args {
        if {$opt ne "-private"} {
            error "tuck: ctx declare_output: unknown option '$opt'"
        }
        set private 1
    }
    set parts [split $relpath /]
    if {$relpath eq "" || "" in $parts || "." in $parts || ".." in $parts} {
        error "tuck: ctx declare_output: invalid path '$relpath'"
    }
    set label [dict get $ctx label]
    set id [::tuck::graph::register_declared $label $relpath]
    if {!$private} {
        dict lappend ::tuck::graph::target_outputs $label $id
    }
    return $id
}
```

```tcl
# Register a static action node for $label. Returns the generated id.
# Errors if any output already has a producing action.
proc ::tuck::graph::register_action {label inputs outputs cmd env} {
    set n 0
    if {[dict exists $::tuck::graph::_action_counter $label]} {
        set n [dict get $::tuck::graph::_action_counter $label]
    }
    set id "$label#$n"
    foreach o $outputs {
        if {[dict exists $::tuck::graph::producers $o]} {
            error "tuck: $o is produced by both [dict get $::tuck::graph::producers $o] and $id"
        }
    }
    dict set ::tuck::graph::_action_counter $label [expr {$n + 1}]
    dict set ::tuck::graph::actions $id [dict create \
        id $id target $label inputs $inputs outputs $outputs \
        cmd $cmd env $env dynamic false]
    foreach o $outputs {
        dict set ::tuck::graph::producers $o $id
    }
    return $id
}
```

```tcl
# The pure-Tcl directory sandbox: run one action in a fresh directory that holds
# only its declared inputs, with a cleared environment. See the M3 spec §6.
#
# Runs in the root interpreter. Changes the process-wide working directory and
# ::env for the duration of the exec; safe only because execution is
# single-threaded.

namespace eval ::tuck::sandbox {
    variable counter 0
}

# Replace the whole environment with dict $d. Unsets elements one by one:
# `array unset ::env` would remove the traces that sync ::env with the process.
proc ::tuck::sandbox::_setenv {d} {
    foreach k [array names ::env] {
        unset ::env($k)
    }
    dict for {k v} $d {
        set ::env($k) $v
    }
}

proc ::tuck::sandbox::_fail {dir keep msg} {
    if {$keep} {
        append msg "\n(sandbox kept at $dir)"
    } else {
        file delete -force $dir
    }
    error $msg
}

# Run action $id. $inputs and $outputs are exec paths relative to $execroot.
# Returns the action's combined stdout+stderr.
proc ::tuck::sandbox::run {id execroot inputs outputs cmd env} {
    variable counter
    set execroot [file normalize $execroot]
    set dir [file join $execroot tuck-out sandbox [incr counter]]
    file delete -force $dir
    file mkdir [file join $dir .home] [file join $dir .tmp]
    foreach p $inputs {
        set link [file join $dir $p]
        file mkdir [file dirname $link]
        file link -symbolic $link [file join $execroot $p]
    }
    foreach p $outputs {
        file mkdir [file dirname [file join $dir $p]]
    }

    set keep [expr {[info exists ::env(TUCK_KEEP_SANDBOX)]
                    && $::env(TUCK_KEEP_SANDBOX) eq "1"}]
    set path [expr {[info exists ::env(PATH)] ? $::env(PATH) : "/usr/bin:/bin"}]
    set clean [dict merge [dict create \
        PATH $path \
        HOME [file join $dir .home] \
        TMPDIR [file join $dir .tmp] \
        LANG C LC_ALL C TZ UTC SOURCE_DATE_EPOCH 0] $env]

    set saved [array get ::env]
```

```tcl
# The execution phase: order the actions needed for some artifacts, decide which
# are stale, and run those in the directory sandbox. See the M3 spec §5–§7.

namespace eval ::tuck::phase {
    variable _order {}
    variable _state {}
}

# Leaves-first list of the action ids needed to produce $artifact_ids.
proc ::tuck::phase::plan_actions {artifact_ids} {
    variable _order {}
    variable _state {}
    foreach aid $artifact_ids {
        _visit_artifact $aid {}
    }
    return $_order
}

proc ::tuck::phase::_visit_artifact {aid path} {
    set art [dict get $::tuck::graph::artifacts $aid]
    if {[dict get $art type] eq "source"} {
        set rel [dict get $art path]
        if {![file isfile [file join $::tuck::config::root $rel]]} {
            error "tuck: missing source file $rel"
        }
        return
    }
    if {![dict exists $::tuck::graph::producers $aid]} {
        error "tuck: no action produces $aid"
    }
    _visit_action [dict get $::tuck::graph::producers $aid] $path
}

# Post-order DFS over input -> producer edges; $path is the current DFS stack,
# used to report cycles.
proc ::tuck::phase::_visit_action {act path} {
    variable _order
    variable _state
    if {[dict exists $_state $act]} {
        if {[dict get $_state $act] eq "visiting"} {
            set cyc [lrange $path [lsearch -exact $path $act] end]
            lappend cyc $act
            error "tuck: action cycle: [join $cyc { -> }]"
        }
        return
    }
    dict set _state $act visiting
    lappend path $act
    foreach in [dict get $::tuck::graph::actions $act inputs] {
        _visit_artifact $in $path
    }
    dict set _state $act done
    lappend _order $act
}
```

```tcl
proc ::tuck::phase::_db_path {} {
    return [file join $::tuck::config::root tuck-out .db]
}

proc ::tuck::phase::_db_load {} {
    set p [_db_path]
    if {![file exists $p]} {
        return {}
    }
    set f [open $p r]
    set d [read $f]
    close $f
    if {[catch {dict size $d}]} {
        return {}   ;# corrupt database: treat everything as stale
    }
    return $d
}

proc ::tuck::phase::_db_save {db} {
    set p [_db_path]
    file mkdir [file dirname $p]
    set f [open $p.tmp w]
    puts -nonewline $f $db
    close $f
    file rename -force $p.tmp $p
}

proc ::tuck::phase::_paths {ids} {
    return [lmap id $ids {::tuck::graph::exec_path $id}]
}

# What, besides its inputs, determines an action's outputs.
proc ::tuck::phase::_key {a} {
    return [list [dict get $a cmd] [dict get $a env]]
}

# exec path -> {mtime size} for each input.
proc ::tuck::phase::_stamps {a} {
    set s {}
    foreach p [_paths [dict get $a inputs]] {
        set f [file join $::tuck::config::root $p]
        dict set s $p [list [file mtime $f] [file size $f]]
    }
    return $s
}

proc ::tuck::phase::_stale {db a} {
    foreach p [_paths [dict get $a outputs]] {
        if {![file exists [file join $::tuck::config::root $p]]} {
            return 1
        }
    }
    set id [dict get $a id]
    if {![dict exists $db $id]} {
        return 1
    }
    set rec [dict get $db $id]
    return [expr {[dict get $rec key] ne [_key $a]
                  || [dict get $rec inputs] ne [_stamps $a]}]
}

proc ::tuck::phase::_report {report args} {
    if {$report ne ""} {
        {*}$report {*}$args
    }
}

# Materialize $artifact_ids. $report, if given, is a command prefix called as
# `{*}$report run|skip <action-id>` and `{*}$report output <action-id> <text>`.
# Returns the list of {run|skip <action-id>} events.
proc ::tuck::phase::execute {artifact_ids {report {}}} {
    set db [_db_load]
    set rebuilt {}
    set events {}
    foreach act [plan_actions $artifact_ids] {
        set a [dict get $::tuck::graph::actions $act]
        set dirty [_stale $db $a]
        # An input rebuilt in this run always makes us stale, even if the new
        # file has the same size and mtime second as the recorded one.
        foreach in [dict get $a inputs] {
            if {[dict exists $::tuck::graph::producers $in]
                && [dict get $::tuck::graph::producers $in] in $rebuilt} {
                set dirty 1
            }
        }
        if {!$dirty} {
            lappend events [list skip $act]
            _report $report skip $act
            continue
        }
        lappend events [list run $act]
        _report $report run $act
        set out [::tuck::sandbox::run $act $::tuck::config::root \
            [_paths [dict get $a inputs]] [_paths [dict get $a outputs]] \
            [dict get $a cmd] [dict get $a env]]
        if {$out ne ""} {
            _report $report output $act $out
        }
        lappend rebuilt $act
        dict set db $act [dict create key [_key $a] inputs [_stamps $a]]
        _db_save $db
    }
    return $events
}
```

```tcl
rules {
    # "a/b/c.h" -> "a/b"; "c.h" -> "c.h"'s directory is "."
    proc dir_of {path} {
        if {[string first / $path] < 0} { return . }
        return [regsub {/[^/]*$} $path {}]
    }

    # "a/b/c.h" -> "c.h"
    proc base_of {path} {
        return [regsub {^.*/} $path {}]
    }

    # Compile each C source to a private object file. $hdrs are header artifacts
    # the compile may include: they become inputs, and their directories -I flags.
    # Returns the object artifact ids.
    proc compile {srcs hdrs copts} {
        set incs {}
        foreach h $hdrs {
            set flag -I[dir_of [ctx path $h]]
            if {$flag ni $incs} { lappend incs $flag }
        }
        set objs {}
        foreach src $srcs {
            set base [regsub {\.c$} [base_of [ctx path $src]] {}]
            set obj [ctx declare_output build/$base.o -private]
            ctx action \
                inputs  [concat [list $src] $hdrs] \
                outputs [list $obj] \
                cmd     [concat [list cc] $copts $incs \
                             [list -c [ctx path $src] -o [ctx path $obj]]]
            lappend objs $obj
        }
        return $objs

    # dependents can compile against them, and archive the objects.
    tuck::rule cc_library {
        attrs {
            name  {type string  mandatory true}
            srcs  {type sources mandatory true}
            hdrs  {type sources mandatory false default {}}
            copts {type list    mandatory false default {}}
        }
        impl {
            set hdrs [ctx attr hdrs]
            set objs [compile [ctx attr srcs] $hdrs [ctx attr copts]]
            foreach h $hdrs {
                set pub [ctx declare_output include/[base_of [ctx path $h]]]
                ctx action inputs [list $h] outputs [list $pub] \
                    cmd [list cp [ctx path $h] [ctx path $pub]]
            }
            set lib [ctx declare_output build/lib[ctx attr name].a]
            set cmd [list ar rcs [ctx path $lib]]
            foreach o $objs { lappend cmd [ctx path $o] }
            ctx action inputs $objs outputs [list $lib] cmd $cmd
        }

    # objects with the deps' archives.
    tuck::rule cc_binary {

            srcs  {type sources mandatory true}
            deps  {type deps    mandatory false default {}}

            ctx action inputs [concat $objs $libs] outputs [list $bin] cmd $cmd
```
