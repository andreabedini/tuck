# Tuck M2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the analysis phase — a `ctx.*` command surface and a driver that runs each target's compiled rule impl leaves-first to produce a static action graph (artifacts + action nodes) — observable via `tuck analyze` and `tuck query actions`.

**Architecture:** Builds directly on M1. Analysis reuses the M1 target node as the `ctx` object, invokes the M1-compiled impl proc (`__tuck_impl_<rule>`) in the shared rules interp, and exposes `ctx.*` commands (aliased into that interp) that read/mutate three new global stores in the root interp: `graph::artifacts`, `graph::actions`, `graph::target_outputs`. No execution and no on-disk paths — `cmd` is stored verbatim.

**Tech Stack:** Tcl 9.0+ (`tclsh`), `tcltest`, Jujutsu (`jj`).

---

## Version control & test conventions

- jj: working copy is a commit; new files auto-tracked; finalize with `jj commit -m "..."`.
- Run one suite: `tclsh tests/<name>.test`; full suite: `tclsh tests/all.tcl`.
- A clean run ends `Total N Passed N Skipped 0 Failed 0`.

## Background the implementer needs

- The **`ctx` object is the M1 target node**: `dict {label rule attrs package deps}`.
- M1 already compiles a rule's inline `impl { ... }` body into a proc
  `__tuck_impl_<name> {ctx} <body>` in the rules interp, and stores that name (or `""`
  if no impl) at `dict get $::tuck::graph::rules <rule> impl`.
- Rule schemas live at `dict get $::tuck::graph::rules <rule> schema`; a schema maps
  `attr -> {type ... mandatory ... default ...}`.
- `::tuck::util::canon <label> <pkg>` canonicalizes a (possibly relative) label.
- `::tuck::phase::load_closure <pkglist>`, `::tuck::graph::validate`,
  `::tuck::graph::topo_sort <seeds>` (leaves-first) already exist from M1.

```tcl
namespace eval ::tuck::graph {
    variable artifacts {}
    variable actions {}
    variable target_outputs {}
    variable _action_counter {}
}

proc ::tuck::graph::source_id {relpath} {
    return "source:$relpath"
}

proc ::tuck::graph::declared_id {label relpath} {
    return "declared:$label:$relpath"
}

# Register a source artifact (idempotent). Returns its id.
proc ::tuck::graph::register_source {relpath} {
    set id [::tuck::graph::source_id $relpath]
    dict set ::tuck::graph::artifacts $id [dict create type source path $relpath]
    return $id
}

# Register a declared artifact (idempotent). Returns its id.
proc ::tuck::graph::register_declared {label relpath} {
    set id [::tuck::graph::declared_id $label $relpath]
    dict set ::tuck::graph::artifacts $id \
        [dict create type declared target $label relpath $relpath]
    return $id
}

# Register a static action node for $label. Returns the generated id.
proc ::tuck::graph::register_action {label inputs outputs cmd env} {
    set n 0
    if {[dict exists $::tuck::graph::_action_counter $label]} {
        set n [dict get $::tuck::graph::_action_counter $label]
    }
    dict set ::tuck::graph::_action_counter $label [expr {$n + 1}]
    set id "$label#$n"
    dict set ::tuck::graph::actions $id [dict create \
        id $id target $label inputs $inputs outputs $outputs \
        cmd $cmd env $env dynamic false]
    return $id
}
```

```tcl
namespace eval ::tuck::ctx {}

# Alias the ctx.* surface into a (rules) interpreter.
proc ::tuck::ctx::install {interp} {
    foreach cmd {attr label package declare_output action dep_outputs} {
        $interp alias ctx.$cmd ::tuck::ctx::$cmd
    }
}

proc ::tuck::ctx::label {ctx} {
    return [dict get $ctx label]
}

proc ::tuck::ctx::package {ctx} {
    return [dict get $ctx package]
}

# repo-relative path for a package-relative source path
proc ::tuck::ctx::_relpath {pkg p} {
    if {$pkg eq ""} {
        return $p
    }
    return "$pkg/$p"
}

# Type-resolved attribute access.
proc ::tuck::ctx::attr {ctx key} {
    set rule   [dict get $ctx rule]
    set schema [dict get $::tuck::graph::rules $rule schema]
    if {![dict exists $schema $key]} {
        error "tuck: ctx.attr: no attribute '$key'"
    }
    set attrs [dict get $ctx attrs]
    if {![dict exists $attrs $key]} {
        return {}
    }
    set type [dict get $schema $key type]
    set val  [dict get $attrs $key]
    set pkg  [dict get $ctx package]
    switch -- $type {
        sources {
            set ids {}
            foreach p $val {
                lappend ids [::tuck::graph::register_source [::tuck::ctx::_relpath $pkg $p]]
            }
            return $ids
        }
        dep {
            return [::tuck::util::canon $val $pkg]
        }
        deps {
            set out {}
            foreach l $val {
                lappend out [::tuck::util::canon $l $pkg]
            }
            return $out
        }
        default {
            return $val
        }
    }
}

# Declare an output artifact for this target; record + return its id.
proc ::tuck::ctx::declare_output {ctx relpath} {
    set label [dict get $ctx label]
    set id [::tuck::graph::register_declared $label $relpath]
    dict lappend ::tuck::graph::target_outputs $label $id
    return $id
}

# Return the declared output ids of an already-analysed dependency.
proc ::tuck::ctx::dep_outputs {ctx label} {
    if {![dict exists $::tuck::graph::target_outputs $label]} {
        error "tuck: ctx.dep_outputs: $label has no analysed outputs"
    }
    return [dict get $::tuck::graph::target_outputs $label]
}

# Register a static action node. Named args: inputs outputs cmd ?env?.
proc ::tuck::ctx::action {ctx args} {
    if {[llength $args] % 2 != 0} {
        error "tuck: ctx.action: arguments must be key/value pairs"
    }
    set spec [dict create env {}]
    foreach {k v} $args {
        dict set spec $k $v
    }
    foreach req {inputs outputs cmd} {
        if {![dict exists $spec $req]} {
            error "tuck: ctx.action: missing $req"
        }
    }
    return [::tuck::graph::register_action [dict get $ctx label] \
        [dict get $spec inputs] [dict get $spec outputs] \
        [dict get $spec cmd] [dict get $spec env]]
}
```

```tcl
namespace eval ::tuck::phase {}

# Analyse the closure of $labels, leaves-first, invoking each target's compiled
# rule impl to populate the action graph.
proc ::tuck::phase::analyse {labels} {
    set canon {}
    foreach l $labels {
        lappend canon [::tuck::util::canon $l ""]
    }
    set pkgs {}
    foreach c $canon {
        lappend pkgs [dict get [::tuck::util::label_parse $c] pkg]
    }
    ::tuck::phase::load_closure $pkgs
    ::tuck::graph::validate
    foreach label [::tuck::graph::topo_sort $canon] {
        set node [dict get $::tuck::graph::targets $label]
        set rule [dict get $node rule]
        set impl [dict get $::tuck::graph::rules $rule impl]
        if {$impl eq ""} {
            error "tuck: rule $rule has no impl"
        }
        $::tuck::interp::rules eval [list $impl $node]
    }
}
```
