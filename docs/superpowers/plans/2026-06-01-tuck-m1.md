# Tuck M1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the loading phase of the Tuck build system — repo-root detection, lazy/transitive `BUILD` loading, target-graph construction with attr validation, label resolution, and topo-sort — exposed via `tuck targets <pattern>` and `tuck query deps <label>`.

**Architecture:** Pure TCL. A root interpreter orchestrates three child interpreters: a short-lived per-file *loader* (exposes only `targets`/`rules`), a shared persistent *target* interp (holds rule constructors), and a shared persistent *rules* interp (holds rule definitions). All child interps are created with `interp create -safe`, which removes `open`/`exec`/`file`/`package` etc. for free — hermeticity comes from the safe-interp boundary, with narrow aliases as the only escape hatch. Rules are repo-global, defined only in the root `TUCK` file, loaded deterministically before any lazy package loading.

**Tech Stack:** Tcl 9.0.2 (`tclsh`), `tcltest` (built-in), Jujutsu (`jj`) for version control.

```tcl
namespace eval ::tuck::config {
    variable root ""
}

# Walk up from $start until a directory containing .tuck/ is found.
proc ::tuck::config::find_root {start} {
    set dir [file normalize $start]
    while 1 {
        if {[file isdirectory [file join $dir .tuck]]} {
            return $dir
        }
        set parent [file dirname $dir]
        if {$parent eq $dir} {
            error "tuck: no .tuck/ found at or above $start"
        }
        set dir $parent
    }
}

# Detect and cache the repo root.
proc ::tuck::config::init {start} {
    variable root
    set root [::tuck::config::find_root $start]
    return $root
}
```

```tcl
namespace eval ::tuck::util {}

proc ::tuck::util::_basename {pkg} {
    if {$pkg eq ""} {
        error "tuck: malformed label: // needs an explicit :name"
    }
    return [lindex [split $pkg /] end]
}

# Parse a label into {pkg <p> name <n>}. Relative ":name" uses $current_pkg.
proc ::tuck::util::label_parse {label {current_pkg ""}} {
    if {[string range $label 0 1] eq "//"} {
        set rest [string range $label 2 end]
        set cidx [string first : $rest]
        if {$cidx < 0} {
            set pkg $rest
            set name [::tuck::util::_basename $pkg]
        } else {
            set pkg  [string range $rest 0 [expr {$cidx - 1}]]
            set name [string range $rest [expr {$cidx + 1}] end]
        }
    } elseif {[string index $label 0] eq ":"} {
        set pkg $current_pkg
        set name [string range $label 1 end]
    } else {
        error "tuck: malformed label: $label"
    }
    if {$name eq ""} {
        error "tuck: malformed label: empty target name in $label"
    }
    return [dict create pkg $pkg name $name]
}

proc ::tuck::util::label_canon {pkg name} {
    return "//$pkg:$name"
}

# Canonicalize an arbitrary (possibly relative) label against $pkg.
proc ::tuck::util::canon {label pkg} {
    set p [::tuck::util::label_parse $label $pkg]
    return [::tuck::util::label_canon [dict get $p pkg] [dict get $p name]]
}
```

```tcl
namespace eval ::tuck {}
namespace eval ::tuck::graph {
    variable rules {}
}

# Type check for a single attr value. Minimal for M1.
proc ::tuck::check_type {type value rule attr} {
    switch -- $type {
        int {
            if {![string is integer -strict $value]} {
                error "tuck: $rule: attribute '$attr' expected int, got '$value'"
            }
        }
        bool {
            if {![string is boolean -strict $value]} {
                error "tuck: $rule: attribute '$attr' expected bool, got '$value'"
            }
        }
        string - list - sources - dep - deps - output {
            # accepted as-is in M1
        }
        default {
            error "tuck: $rule: attribute '$attr' has unknown type '$type'"
        }
    }
}

# Validate an attrs dict against a schema. Returns the validated dict with
# defaults applied. Errors on missing-mandatory, unknown-attr, bad-type.
proc ::tuck::validate_attrs {rule schema attrs} {
    set out {}
    dict for {attr def} $schema {
        set mandatory [expr {[dict exists $def mandatory] && [dict get $def mandatory]}]
        if {[dict exists $attrs $attr]} {
            set val [dict get $attrs $attr]
            ::tuck::check_type [dict get $def type] $val $rule $attr
            dict set out $attr $val
        } elseif {$mandatory} {
            error "tuck: $rule: missing mandatory attribute '$attr'"
        } elseif {[dict exists $def default]} {
            dict set out $attr [dict get $def default]
        }
    }
    dict for {k _} $attrs {
        if {![dict exists $schema $k]} {
            error "tuck: $rule: unknown attribute '$k'"
        }
    }
    return $out
}

# Register a rule: store schema+impl, and install its constructor.
proc ::tuck::register_rule {name body} {
    variable graph::rules
    set schema [dict get $body attrs]
    set impl   [expr {[dict exists $body impl] ? [dict get $body impl] : ""}]
    dict set ::tuck::graph::rules $name [dict create schema $schema impl $impl]
    ::tuck::_install_ctor $name
}
```

Note: `::tuck::_install_ctor` is defined in Task 6 (the loader, which owns the target interp). The Task 3 test stubs it.

```tcl
namespace eval ::tuck {
    variable _current_pkg ""
}
namespace eval ::tuck::graph {
    variable targets {}
}

# Collect dep labels from attrs whose schema type is dep/deps, canonicalized.
proc ::tuck::extract_deps {schema attrs pkg} {
    set deps {}
    dict for {attr def} $schema {
        if {![dict exists $attrs $attr]} continue
        switch -- [dict get $def type] {
            dep {
                lappend deps [::tuck::util::canon [dict get $attrs $attr] $pkg]
            }
            deps {
                foreach l [dict get $attrs $attr] {
                    lappend deps [::tuck::util::canon $l $pkg]
                }
            }
        }
    }
    return $deps
}

# Constructor callback: validate attrs, build node, register it.
proc ::tuck::register_target {rule attrs} {
    variable _current_pkg
    set rdef   [dict get $::tuck::graph::rules $rule]
    set schema [dict get $rdef schema]
    set valid  [::tuck::validate_attrs $rule $schema $attrs]
    set name   [dict get $valid name]
    set label  [::tuck::util::label_canon $_current_pkg $name]
    if {[dict exists $::tuck::graph::targets $label]} {
        error "tuck: duplicate target label: $label"
    }
    set deps [::tuck::extract_deps $schema $valid $_current_pkg]
    dict set ::tuck::graph::targets $label [dict create \
        label $label rule $rule attrs $valid package $_current_pkg deps $deps]
}
```

```tcl
namespace eval ::tuck::graph {
    variable targets {}
}

# Every dep of every loaded target must name a known target.
proc ::tuck::graph::validate {} {
    variable targets
    dict for {label node} $targets {
        foreach dep [dict get $node deps] {
            if {![dict exists $targets $dep]} {
                error "tuck: $label depends on unknown target $dep"
            }
        }
    }
    return ""
}

proc ::tuck::graph::_visit {label stateVar resultVar path} {
    variable targets
    upvar 1 $stateVar state $resultVar result
    if {[dict exists $state $label]} {
        if {[dict get $state $label] eq "visiting"} {
            error "tuck: dependency cycle: [join [concat $path $label] { -> }]"
        }
        return
    }
    dict set state $label visiting
    foreach dep [dict get $targets $label deps] {
        ::tuck::graph::_visit $dep state result [concat $path $label]
    }
    dict set state $label done
    lappend result $label
}

# Post-order DFS topo sort from $seeds. Leaves first. Detects cycles.
proc ::tuck::graph::topo_sort {seeds} {
    set state {}
    set result {}
    foreach s $seeds {
        ::tuck::graph::_visit $s state result {}
    }
    return $result
}
```

```tcl
namespace eval ::tuck::phase {}
namespace eval ::tuck::interp {
    variable target ""
    variable rules  ""
}

# Reset all mutable state — used between test cases and at process start.
proc ::tuck::reset {} {
    set ::tuck::graph::rules {}
    set ::tuck::graph::targets {}
    set ::tuck::_current_pkg ""
    if {$::tuck::interp::target ne "" && [interp exists $::tuck::interp::target]} {
        interp delete $::tuck::interp::target
    }
    if {$::tuck::interp::rules ne "" && [interp exists $::tuck::interp::rules]} {
        interp delete $::tuck::interp::rules
    }
    set ::tuck::interp::target ""
    set ::tuck::interp::rules ""
}

# Create the shared, persistent target and rules interpreters.
proc ::tuck::phase::setup {} {
    set ::tuck::interp::target [interp create -safe]
    set ::tuck::interp::rules  [interp create -safe]
    $::tuck::interp::rules alias tuck::rule ::tuck::register_rule
}

# Install a rule constructor (named $name) into the shared target interp.
# Called by ::tuck::register_rule.
proc ::tuck::_install_ctor {name} {
    $::tuck::interp::target alias $name ::tuck::register_target $name
}

# Read a file from disk (root interp only) and return its contents.
proc ::tuck::phase::_slurp {path} {
    set fh [open $path r]
    try {
        return [read $fh]
    } finally {
        close $fh
    }
}

# Evaluate one BUILD/TUCK file for package $pkg inside a fresh loader interp.
proc ::tuck::phase::load_file {path pkg} {
    set ::tuck::_current_pkg $pkg
    set src [::tuck::phase::_slurp $path]
    set loader [interp create -safe]
    $loader alias targets ::tuck::phase::_eval_targets
    $loader alias rules   ::tuck::phase::_eval_rules
    try {
        $loader eval $src
    } finally {
        interp delete $loader
    }
}

proc ::tuck::phase::_eval_targets {body} {
    $::tuck::interp::target eval $body
}
proc ::tuck::phase::_eval_rules {body} {
    $::tuck::interp::rules eval $body
}

# Load the root TUCK file (rules only), if present. Deterministic, first.
proc ::tuck::phase::load_root_rules {} {
    set tuckfile [file join $::tuck::config::root TUCK]
    if {[file exists $tuckfile]} {
        ::tuck::phase::load_file $tuckfile ""
    }
}

# Load a single package's BUILD file.
proc ::tuck::phase::load_package {pkg} {
    set bf [file join $::tuck::config::root $pkg BUILD]
    if {![file exists $bf]} {
        error "tuck: no BUILD file for package //$pkg"
    }
    ::tuck::phase::load_file $bf $pkg
}

# Lazily load $seed_pkgs and the transitive closure of their dep packages.
proc ::tuck::phase::load_closure {seed_pkgs} {
    set loaded {}
    set pending $seed_pkgs
    while {[llength $pending]} {
        set pkg [lindex $pending 0]
        set pending [lrange $pending 1 end]
        if {[dict exists $loaded $pkg]} continue
        dict set loaded $pkg 1
        ::tuck::phase::load_package $pkg
        dict for {label node} $::tuck::graph::targets {
            foreach dep [dict get $node deps] {
                set dpkg [dict get [::tuck::util::label_parse $dep] pkg]
                if {![dict exists $loaded $dpkg]} {
                    lappend pending $dpkg
                }
            }
        }
    }
}
```

```tcl
# Find every package (dir containing BUILD) at or under $subdir, relative to root.
proc ::tuck::cmd::_scan_packages {subdir} {
    set base [file join $::tuck::config::root $subdir]
    set pkgs {}
    foreach bf [::tuck::cmd::_find_builds $base] {
        set rel [::tuck::cmd::_relpath $::tuck::config::root [file dirname $bf]]
        lappend pkgs $rel
    }
    return $pkgs
}

proc ::tuck::cmd::_find_builds {dir} {
    set found {}
    if {[file exists [file join $dir BUILD]]} {
        lappend found [file join $dir BUILD]
    }
    foreach sub [glob -nocomplain -types d -directory $dir *] {
        if {[file tail $sub] eq ".tuck"} continue
        lappend found {*}[::tuck::cmd::_find_builds $sub]
    }
    return $found
}

proc ::tuck::cmd::_relpath {root path} {
    set r [file split [file normalize $root]]
    set p [file split [file normalize $path]]
    set tail [lrange $p [llength $r] end]
    return [join $tail /]
}

proc ::tuck::cmd::targets {pattern} {
    if {$pattern eq "//..." || [string match "*/..." $pattern]} {
        set sub [string range $pattern 2 end]      ;# strip //
        set sub [string range $sub 0 end-3]        ;# strip "..."
        set sub [string trimright $sub /]          ;# strip trailing /
        set pkgs [::tuck::cmd::_scan_packages $sub]
        foreach pkg $pkgs { ::tuck::phase::load_package $pkg }
        ::tuck::graph::validate
        ::tuck::cmd::_print_all
    } elseif {[string match "*:..." $pattern]} {
        set pkg [string range $pattern 2 end-4]    ;# strip // and ":..."
        ::tuck::phase::load_package $pkg
        ::tuck::cmd::_print_pkg $pkg
    } else {
        set canon [::tuck::util::canon $pattern ""]
        set pkg [dict get [::tuck::util::label_parse $canon] pkg]
        ::tuck::phase::load_package $pkg
        if {![dict exists $::tuck::graph::targets $canon]} {
            error "tuck: no such target: $canon"
        }
        puts $canon
    }
}

proc ::tuck::cmd::_print_all {} {
    foreach l [lsort [dict keys $::tuck::graph::targets]] { puts $l }
}

proc ::tuck::cmd::_print_pkg {pkg} {
    set out {}
    dict for {l node} $::tuck::graph::targets {
        if {[dict get $node package] eq $pkg} { lappend out $l }
    }
    foreach l [lsort $out] { puts $l }
}
```
